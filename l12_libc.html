IMPORT(hello)        // ? I need hello

EXPORT(_start)       # I give _start

      
main:

      
    call DSYM(hello)  // = call *hello(%rip)

      
    …                 // four times

      
_start:

      
    call main

      
    jmp DSYM(exit)
