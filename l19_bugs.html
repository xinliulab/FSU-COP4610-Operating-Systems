DeadlockAA: one thread takes its own lock twice

    
aa-deadlock.c

      
mutex_t lock = MUTEX_INIT();

      
void T_lock(void *arg) {

      
    mutex_lock(&lock);

      
    printf("Thread %ld acquired lock\n", (long)arg);

      
    // the same thread takes the same lock again

      
    mutex_lock(&lock);

      
    printf("Thread %ld acquired lock again"

      
           " (never reached)\n", (long)arg);

      
    mutex_unlock(&lock);

      
    mutex_unlock(&lock);

      
}

      
int main() { spawn(T_lock); }

BreakFour conditions, all needed

    
1 · Mutual exclusionone key, one holder

    
2 · Hold and waita holder wants more keys

    
3 · No preemptionnobody can take a key away

    
4 · Circular waitthe waits close a loop

    

    

    
All four must hold at once; break any one and no deadlock can form

    
E. G. Coffman, M. J. Elphick, A. Shoshani, "System Deadlocks", ACM Computing Surveys 3(2), 1971 · the keys-on-a-table reading follows the source lecture

OrderRun lockdep on the philosophers

    
lockdep.c

      
int lockdep_mutex_lock(pthread_mutex_t *m) {

      
    int id = register_mutex(m);

      
    for (int i = 0; i < nheld; i++) {

      
        int prev = held[i];

      
        int visited[MAX_LOCKS] = {};

      
        if (has_path(id, prev, visited))

      
            fprintf(stderr, "*** LOCKDEP: "

      
                "potential deadlock! ***");

      
        edge[prev][id] = 1;

      
    }

      
    held[nheld++] = id;

      
    return real_lock(m);

      
}

RaceYour races: the wrong lock, or no lock

    
race.c

      
long sum = 0;

      
mutex_t lk = MUTEX_INIT();

      
void T_sum() {             // two threads

      
    for (int i = 0; i < N; i++) {   // N = 10^6

      
        mutex_lock(&lk);

      
        sum++;

      
        mutex_unlock(&lk);

      
        if (i % (N / 10) == 0) {

      
            sum++;     // no lock

      
        }

      
    }

      
}

DebugA two-line lockdep: the spin limit

    
spin-limit.c

      
#define SPIN_LIMIT 100000000

      
void spin_lock_at(spinlock_t *lk, const char *file, int line) {

      
    long spin_cnt = 0;

      
    while (__atomic_exchange_n(lk, 1, __ATOMIC_ACQUIRE) == 1) {

      
        if (spin_cnt++ > SPIN_LIMIT)

      
            panic("Spin limit exceeded @ %s:%d\n", file, line);

      
    }

      
}

      
#define spin_lock(lk) spin_lock_at(lk, __FILE__, __LINE__)

      
 

      
void T_lock() {

      
    spin_lock(&lk); printf("acquired lk\n");

      
    spin_lock(&lk);    // line 30: AA, the same lock again

      
}
