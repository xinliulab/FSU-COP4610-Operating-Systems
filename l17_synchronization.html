}

Allowed: only <><_ and ><>_ (fish)

    
    

      
// thread: fish_before(c); putchar(c); fish_after(c);

      
void fish_before(char ch) {

      
    mutex_lock(&lk);

      
    while (!can_print(ch)) {  // an edge for ch, quota > 0

      
        cond_wait(&cv, &lk);

      
    }

      
    quota--;

      
    mutex_unlock(&lk);

      
}

      
void fish_after(char ch) {

      
    mutex_lock(&lk);

      
    quota++;

      
    current = next(ch);       // follow the edge

      
    cond_broadcast(&cv);

      
    mutex_unlock(&lk);

      
}

void add_atom(char a) {     // 4 H, 2 O threads

      
    mutex_lock(&lk);

      
    while (!(a == 'H' ? h < 2 : o < 1)) {

      
        cond_wait(&cv, &lk);

      
    }

      
    putchar(a);

      
    if (a == 'H') h++; else o++;

      
    if (h == 2 && o == 1) {      // one molecule done

      
        putchar(' ');

      
        h = o = 0;

      
    }

      
    cond_broadcast(&cv);

      
    mutex_unlock(&lk);

      
}

Computation graphsNodes do the work, edges wait

    
    
node: a piece of workit may read and write shared memory

    
edge u → v: v uses u's resultv starts after u ends, and sees all u's writes

    
With 1 s per node: 4 s, not 6 snodes 1, 2, 3 run at the same time

make: the build graph

    
Any parallel program is a graph; every edge is one wait-until

void T_worker(int tid) {

      
  mutex_lock(&mutex);

      
  while (!(tasks[tid].pending_deps == 0))

      
    cond_wait(&tasks[tid].cv, &mutex);

      
  mutex_unlock(&mutex);

      
  printf("Computing node #%d...\n", tid);

      
  sleep(1);                  // the work

      
  mutex_lock(&mutex);

      
  for (int i = 0; i < LENGTH(edges); i++) {

      
    struct task *t = &tasks[edges[i].dst];

      
    if (edges[i].src == tid) {

      
      t->pending_deps--;

      
      if (t->pending_deps == 0)

      
        cond_broadcast(&t->cv);

      
    }

      
  }

      
  mutex_unlock(&mutex);

      
}
