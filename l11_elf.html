LoadA real ELF loader: one loop of mmap

    

      
Elf64_Ehdr *h = mmap(NULL, 4096, PROT_READ, MAP_PRIVATE, fd, 0);

      
Elf64_Phdr *pht = (Elf64_Phdr *)((char *)h + h->e_phoff);

      
for (int i = 0; i < h->e_phnum; i++) {

      
  Elf64_Phdr *p = &pht[i];

      
  if (p->p_type == PT_LOAD) {

      
    if (p->p_flags & PF_R) prot |= PROT_READ;   // W, X too

      
    mmap((void *)map_beg, map_sz, prot,

      
         MAP_PRIVATE | MAP_FIXED, fd, ROUND(p->p_offset, align));

      
  }

      
}

      
void *rsp = init_proc_stack(argv, envp);
