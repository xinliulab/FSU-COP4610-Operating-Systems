static ssize_t launcher_read(struct file *file,

      
        char __user *buf, size_t count, loff_t *offset) {

      
    if (*offset != 0) {

      
        return 0;

      
    } else {

      
        const char *data = "This is dangerous!\n";

      
        size_t datalen = strlen(data);

      
        if (count > datalen) {

      
          count = datalen;

      
        }

      
        if (copy_to_user(buf, data, count)) {

      
          return -EFAULT;

      
        }

      
        *offset += count;

      
        return count;

      
    }

      
}
